# 配置字段中文说明

本文集中保存字段讲解，不是另一套运行配置。实际填写 `.tcode-local/`，初始化源为 `config/templates/`。示例值不代表正式模板默认值。JSON 不支持注释，请勿直接复制整段作为运行文件。

## human_review

```jsonc
{
  // 总开关。false 时不按本文件拦截 MCP 工具。
  "enabled": true,

  // 当前运行模式标识，供接口和前端展示。
  "mode": "hybrid",

  // 以 MCP 服务名为键，登记需要审核的工具。
  "required_review_tools": {
    "gnina-mcp": {
      // 工具名必须与 MCP tools/list 返回的名称完全一致。
      "tools": ["gnina_mcp_server"],
      // 展示给审核人的触发原因。
      "reason": "分子对接结果需要人工确认",
      // 参数命中任一条件时触发；空对象表示每次调用都触发。
      "auto_review_conditions": {
        "num_modes": {
          // 支持 gt、lt、gte、lte、eq 和 neq。
          "operator": "gt",
          "value": 10
        }
      }
    }
  },

  // Web 审核接口使用的通用自动检查规则。
  "auto_review_rules": {
    "file_modifications": {
      "enabled": true,
      // 正则表达式匹配修改文件名。
      "patterns": [".*\\.py$", ".*\\.ts$", ".*\\.js$"],
      "max_modified_files": 5
    },
    "tool_output_threshold": {
      "enabled": true,
      "max_results": 10,
      "confidence_threshold": 0.8
    },
    "user_request_keywords": {
      "enabled": true,
      "keywords": ["review", "check", "verify", "confirm", "审核", "确认", "检查"]
    }
  },

  // 审核记录的状态机和默认超时。
  "review_workflow": {
    "states": ["pending_review", "approved", "rejected", "needs_revision"],
    "default_state": "pending_review",
    "auto_approve_labels": ["docs", "test", "trivial"],
    "timeout_hours": 24
  },

  // 预留通知配置；当前为空时不会向外部渠道发送消息。
  "notification": {
    "enabled": false,
    "channels": []
  }
}
```

## mcp

```jsonc
{
  // 顶层 key 是稳定的 MCP 服务 ID，会用于工具路由和 Agent 可见范围配置。
  "my-mcp": {
    // 当前远端协议使用 streamable-http。
    "transport": "streamable-http",
    // MCP HTTP 入口。公开模板只能使用示例地址。
    "url": "https://mcp.example.com/mcp",
    // 建立会话和读取工具列表的超时，单位为毫秒。
    "connect_timeout": 10000,
    // 该服务的默认工具调用超时，单位为毫秒。
    "timeout": 300000,
    // 后台计算总等待时限，0 为不设总时限；不关闭上面的连接超时。单位为秒。
    "tool_timeout_seconds": {
      "long_running_tool": 0
    },
    // 可选 HTTP 请求头。${ENV_NAME} 会在连接时从环境变量读取，缺失时拒绝连接。
    "headers": {
      "Authorization": "Bearer ${TCODE_MCP_TOKEN}"
    },
    // 总开关。完成 URL、认证和权限检查后再设为 true。
    "enabled": false
  }
}
```

## mcp_agent_config

```jsonc
{
  // 给管理员看的配置说明，不参与权限判断。
  "description": "Agent-specific MCP visibility",
  // key 是 Agent ID；每个 Agent 只看见这里允许且在 mcp.json 中已启用的服务。
  "agents": {
    "build": {
      // 允许直接暴露给该 Agent 的 MCP 服务 ID。
      "allowed_tools": ["job-protocol-mcp"],
      // 允许通过任务路由器访问的服务范围；all 表示仍由路由策略继续收窄。
      "allowed_routed_servers": "all",
      // 给管理员看的职责说明，不参与权限判断。
      "description": "默认 Agent 只直接使用任务协议。"
    }
  }
}
```

## plugins

```jsonc
{
  // 是否扫描允许的插件目录并自动登记。公开和多人部署模板建议保持 false。
  "auto_discover": false,
  // 管理员审核并登记的插件列表。
  "plugins": [
    {
      // 插件稳定 ID，必须与 plugin.json 和代码中的 PluginMetadata 一致。
      "name": "example-plugin",
      // 插件目录；相对路径以配置/工作目录解析，并受允许根目录约束。
      "path": "./plugins/example_plugin",
      // 是否加载插件。审核 Manifest、权限、依赖和执行模式后再开启。
      "enabled": false,
      // 传给插件的配置，具体字段由插件自己的 configSchema 定义。
      "config": {
        "default_prefix": "Echo"
      },
      // 配置来源标签，用于管理界面和诊断信息。
      "source": "plugins.json"
    }
  ]
}
```

## sandbox_runner_tool_policy

```jsonc
{
  // 顶层 key 是工具名。Runner 收到请求后会用此处规则再次校验，
  // Web 或 Agent 传入的参数不能扩大这里授予的范围。
  "browser": {
    // 该工具允许使用的容器镜像白名单。
    "images": ["tcode-browser:playwright"],
    // 允许的网络模式：none、internal 或 egress。
    "networks": ["egress"],
    // true 表示容器根文件系统必须只读。
    "require_read_only": true,
    // false 表示拒绝以容器 root 身份运行。
    "allow_root": false
  },

  "bash": {
    "images": ["tcode-executor:py312", "tcode-tools-blast:latest"],
    "networks": ["none", "egress"],
    // true 只表示该工具具备申请 root 的上限；用户白名单和 Sandbox Profile 仍会继续校验。
    "allow_root": true
  },

  // 不需要网络的插件运行时应使用独立规则和只读根文件系统。
  "plugin_runtime": {
    "images": ["tcode-executor:py312"],
    "networks": ["none"],
    "require_read_only": true,
    "allow_root": false
  },

  // 确实需要外网的插件必须显式走 plugin_network 能力。
  "plugin_network": {
    "images": ["tcode-executor:py312"],
    "networks": ["egress"],
    "require_read_only": true,
    "allow_root": false
  },

  // “工具名#能力名”可以覆盖同一工具的默认规则。
  "skill_run#bio.blast.local": {
    "images": ["tcode-tools-blast:latest"],
    "networks": ["none", "egress"]
  }
}
```

## tcode

```jsonc
{
  // 编辑器用它检查字段类型并显示中文说明；Schema 始终保留在公开 config/ 目录。
  "$schema": "../tcode.schema.json",

  // 默认文本模型，格式固定为“provider ID/模型 ID”。
  "model": "demo/your-model-id",

  // 默认视觉模型；不需要图片输入时可以删除这一项。
  "vision_model": "demo/your-vision-model-id",

  // 模型调用运行参数。
  "runtime": {
    // 单次模型请求的超时秒数。
    "llm_timeout_seconds": 120
  },

  // 内置命令和文件工具的宿主策略。多人环境还需要配置独立沙箱执行器。
  "command_policy": {
    // enforce 表示实际执行策略；audit 只记录策略判定，不能用于生产隔离。
    "mode": "enforce",
    // 是否允许写入当前工作区。
    "allow_workspace_write": true,
    // 是否允许工具访问网络。
    "allow_network": false,
    // 是否允许工具安装依赖。
    "allow_dependency_install": false,
    // 是否允许系统级操作。
    "allow_system_operations": false,
    // 是否允许启动后台进程。
    "allow_background_process": false,
    // 是否允许读取工作区外的文件。
    "allow_external_read": false,
    // 是否允许写入工作区外的文件。
    "allow_external_write": false
  },

  // 从成功任务提炼反思和提示的本地学习功能。
  "learning": {
    // 学习功能总开关。
    "enabled": true,
    "reflection": {
      // 是否记录任务反思。
      "enabled": true,
      // 反思记录保留天数。
      "ttl_days": 30,
      // 每个作用域最多保留的反思数量。
      "max_per_scope": 400,
      // 至少发生多少次工具调用才生成反思。
      "min_tool_calls": 1,
      // 回复至少达到多少字符才生成反思。
      "min_response_chars": 80
    },
    "compaction_flush": {
      // 上下文压缩时是否把可复用信息写入长期存储。
      "enabled": true,
      // 压缩产生的临时学习记录保留天数。
      "ttl_days": 21
    },
    "skill_drafts": {
      // 是否根据重复成功的任务生成技能草稿。
      "enabled": true,
      // 相似任务至少成功多少次才允许生成草稿。
      "min_success_count": 2,
      // 任务至少使用多少个工具才允许生成草稿。
      "min_tool_count": 2
    },
    "adaptive_hints": {
      // 是否把高置信度历史经验加入后续上下文。
      "enabled": true,
      // 经验进入上下文所需的最低置信度。
      "min_confidence": 0.55,
      // 单次最多注入的反思数量。
      "max_reflections": 2,
      // 单次最多注入的提示数量。
      "max_hints": 3,
      // 只检索最近多少天的经验。
      "window_days": 45
    }
  },

  // 模型服务列表。provider ID 是下面的 demo，也是 model 前缀。
  "provider": {
    "demo": {
      // 当前兼容层类型；OpenAI-compatible 服务使用这个值。
      "npm": "@ai-sdk/openai-compatible",
      // 展示用的默认模型名称。
      "name": "your-model-id",
      "options": {
        // 从这个环境变量读取 API 基础地址，避免把部署地址写进仓库。
        "baseURLEnv": "TCODE_DEMO_BASE_URL",
        // 从这个环境变量读取密钥，避免明文凭据进入配置文件。
        "apiKeyEnv": "TCODE_DEMO_API_KEY",
        // 该 provider 的请求超时秒数。
        "timeoutSeconds": 120,
        // OpenAI 系接口协议：chat_completions（默认）/ responses / auto（优先 responses，端点不存在时回退）。
        "apiMode": "chat_completions"
      },
      "models": {
        "your-model-id": {
          // Web 界面显示名称。
          "name": "My text model",
          // 是否接受图片输入。
          "supports_images": false,
          // 明确声明模型接受的输入类型。
          "input_modalities": ["text"]
        },
        "your-vision-model-id": {
          "name": "My vision model",
          "supports_images": true,
          "input_modalities": ["text", "image"]
        }
      }
    }
  },

  // 向量嵌入功能；关闭时基础聊天不依赖本地或远程嵌入服务。
  "embeddings": {
    // 是否启用向量检索。
    "enabled": false,
    // 嵌入服务类型。
    "provider": "ollama",
    // 嵌入模型名称。
    "model": "your-embedding-model",
    "options": {
      // 嵌入服务地址；正式部署建议改用 baseURLEnv 与 apiKeyEnv。
      "baseURL": "http://127.0.0.1:11434/v1"
    }
  }
}
```
